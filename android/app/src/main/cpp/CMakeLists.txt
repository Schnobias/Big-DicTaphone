cmake_minimum_required(VERSION 3.22.1)
project(bigdictaphone LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-O3)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(WHISPER_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../whisper.cpp)
if(NOT EXISTS "${WHISPER_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Run git submodule update --init --recursive before building")
endif()
add_subdirectory(${WHISPER_ROOT} whisper)
add_library(dictaphone-whisper SHARED whisper_jni.cpp)
target_link_libraries(dictaphone-whisper whisper android log)
target_link_options(dictaphone-whisper PRIVATE "-Wl,-z,max-page-size=16384")
